function startSensorSimulation() {

  if (sensorInterval) clearInterval(sensorInterval);

  const tick = () => {

    sensorState.moisture = clamp(sensorState.moisture + rnd(-3,3), 10, 100);

    sensorState.temp = clamp(sensorState.temp + rnd(-1,1), 5, 45);

    sensorState.humidity = clamp(sensorState.humidity + rnd(-2,2), 20, 95);

    sensorState.ec = +clamp(sensorState.ec + rnd(-0.03,0.03), 0.1, 2.0).toFixed(2);


    document.getElementById('sensorMoisture').textContent = Math.round(sensorState.moisture);

    document.getElementById('sensorTemp').textContent = Math.round(sensorState.temp);

    document.getElementById('sensorHumidity').textContent = Math.round(sensorState.humidity);

    document.getElementById('sensorEC').textContent = sensorState.ec;

    document.getElementById('moistureBar').style.width = sensorState.moisture + '%';


    // AI Decision

    let msg = '', cls = 'info';

    const pkg = currentUser.package;

    if ((pkg==='silver'||pkg==='gold') && autoModeEnabled) {

      if (sensorState.moisture < 30 && !irrigationActive) {

        msg = '🚰 AI Otonom: Sulama başlatıldı (nem < 30%)';

        cls = 'warning'; irrigationActive = true;

        addActivity('🤖 AI otonom sulama başlattı'); kpiStats.water += 250;

        setTimeout(() => { sensorState.moisture = Math.min(100, sensorState.moisture+10); },500);

      } else if (sensorState.moisture > 70 && irrigationActive) {

        msg = '✅ AI Otonom: Sulama durduruldu (nem ideal)';

        cls = 'success'; irrigationActive = false;

        addActivity('🤖 AI otonom sulama durdurdu');

      } else if (irrigationActive) msg = '💧 Otonom sulama devam ediyor...';

      else if (sensorState.moisture > 70) msg = '✅ Nem ideal. Sulama gerekmez.';

      else msg = `📊 İzleme: Nem %${Math.round(sensorState.moisture)}, Sıcaklık ${Math.round(sensorState.temp)}°C`;

    } else {

      if (sensorState.moisture < 25) { msg = '⚠️ Kritik: Nem çok düşük! Manuel sulama önerilir.'; cls = 'warning'; }

      else if (sensorState.moisture > 85) { msg = '⚠️ Uyarı: Aşırı nem! Drenaj kontrol edin.'; cls = 'warning'; }

      else msg = `📡 İzleme modu aktif | Nem: %${Math.round(sensorState.moisture)} | Sıcaklık: ${Math.round(sensorState.temp)}°C`;

    }

    const dec = document.getElementById('aiDecision');

    dec.innerHTML = '🤖 AI Kararı: ' + msg;

    dec.className = 'alert alert-' + cls;


    // Chart update

    if (farmerChart) {

      const t = new Date().toLocaleTimeString('tr-TR', {hour:'2-digit',minute:'2-digit',second:'2-digit'});

      farmerChart.data.labels.push(t);

      farmerChart.data.datasets[0].data.push(Math.round(sensorState.moisture));

      farmerChart.data.datasets[1].data.push(Math.round(sensorState.temp));

      farmerChart.data.datasets[2].data.push(Math.round(sensorState.humidity));

      if (farmerChart.data.labels.length > 24) {

        farmerChart.data.labels.shift();

        farmerChart.data.datasets.forEach(d => d.data.shift());

      }

      farmerChart.update('none');

    }


    // KPI update

    kpiStats.co2 = Math.round(kpiStats.water * 0.0003);

    kpiStats.fertilizer = Math.round(kpiStats.sessions * 0.12);

    if(document.getElementById('kpiWater')) {

      document.getElementById('kpiWater').textContent = kpiStats.water;

      document.getElementById('kpiCO2').textContent = kpiStats.co2;

      document.getElementById('kpiFertilizer').textContent = kpiStats.fertilizer;

      document.getElementById('kpiSession').textContent = kpiStats.sessions;

    }


    // Quota

    if (currentUser.quota > 0) { currentUser.quota--; document.getElementById('farmerQuota').textContent = currentUser.quota; DB.updateFarmer(currentUser); }

    kpiStats.sessions++;

  };

  sensorInterval = setInterval(tick, 4000);

  tick();

}

function placeSensors(orderId) {

  const orders=DB.orders();const order=orders.find(o=>o.id===orderId);if(!order)return;

  const count=prompt(`${order.sensorCount} sensör için kurulum onaylansın mı? Kurulacak adet:`,order.sensorCount);

  if(!count||isNaN(count))return;

  order.status='completed'; DB.set('orders',orders);

  let sensors=DB.sensors();

  for(let i=0;i<parseInt(count);i++){

    sensors.push({id:'s'+Date.now()+i,name:`Sensör-${sensors.length+1}`,farmerId:order.farmerId,parcelId:order.parcelId,status:'Aktif',lat:37+Math.random()*5,lng:29+Math.random()*8,installedAt:new Date().toISOString()});

  }

  DB.set('sensors',sensors);

  const fm=DB.get('field');const fi=fm.findIndex(f=>f.id===currentUser.id);

  if(fi!==-1){if(!fm[fi].history)fm[fi].history=[];fm[fi].history.unshift({action:`${count} sensör kurulumu — ${order.farmerId}`,orderId,date:new Date().toISOString()});DB.set('field',fm);}

  renderPendingOrders();renderFieldStats();renderFieldHistory();

  showToast(`${count} sensör kurulumu tamamlandı`,'success');

}
